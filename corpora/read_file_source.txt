43	/**
44	 * Orchestrates TCGPlayer price ingestion across all cards in the database.
45	 *
46	 * Each card's price snapshot is fetched at most once per day (deduped by
47	 * card ID + date). A configurable daily token budget caps total API spend.
48	 *
49	 * Supports tiered ingest:
50	 * - daily-tier sets are fetched every run (typically ~22 sets)
51	 * - weekly-tier sets are spread across 7 days using day-of-week bucketing
52	 *   sorted by card count DESC so large and small sets are interleaved evenly
53	 * - A 7-day fallback skip prevents re-fetching weekly sets that were recently processed
54	 */
55	@Service
56	public class PriceIngestService {
57	
58	    private static final Logger log = LoggerFactory.getLogger(PriceIngestService.class);
59	
60	    /** Credits consumed per TCGPlayer API call: 1 (basic) or 2 (basic + history) */
61	    private static final int TCG_CREDITS_BASIC = 1;
62	    private static final int TCG_CREDITS_WITH_HISTORY = 2;
63	
64	    @Value("${yawn.price-ingest.daily-token-budget:20000}")
65	    private int dailyTokenBudget;
66	
67	    @Value("${yawn.price-ingest.history-card-count:5000}")
68	    private int historyCardCount;
69	
70	    /** Seconds to wait between processing sets during batch ingests. */
71	    @Value("${yawn.price-ingest.inter-set-delay-sec:3}")
72	    private int interSetDelaySec;
73	
74	    /** Seconds to wait before retrying a set after a 429. Multiplied by retry attempt number. */
75	    @Value("${yawn.price-ingest.rate-limit-retry-wait-sec:65}")
76	    private int rateLimitRetryWaitSec;
77	
78	    /** Max number of times to retry a single set after a 429 before stopping the batch. */
79	    @Value("${yawn.price-ingest.rate-limit-max-retries:2}")
80	    private int rateLimitMaxRetries;
81	
82	    /** Days to skip a weekly-tier set if it was already fetched recently. */
83	    @Value("${yawn.price-ingest.weekly-skip-days:7}")
84	    private int weeklySkipDays;
85	
86	    /** Maximum age in days for retry queue entries before they are purged. */
87	    @Value("${yawn.price-ingest.retry-queue-max-age-days:7}")
88	    private int retryQueueMaxAgeDays;
89
